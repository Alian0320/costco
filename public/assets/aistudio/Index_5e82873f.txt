        const btnOnline = document.getElementById('tab-online');
        const btnPickup = document.getElementById('tab-pickup');
        const btnNotice = document.getElementById('tab-notice');
        const btnOrders = document.getElementById('tab-orders');
        const btnHistory = document.getElementById('tab-history');
        const btnProductDb = document.getElementById('tab-product-db');

        const secOnline = document.getElementById('section-online');
        const secPickup = document.getElementById('section-pickup');

        const secHistory = document.getElementById('section-history');
        const secProductDb = document.getElementById('section-product-db');

        btnPurchase.className = 'nav-link fw-bold px-3 ' + (tab === 'purchase' ? 'active' : 'text-secondary');
        btnOnline.className = 'nav-link fw-bold px-3 ' + (tab === 'online' ? 'active' : 'text-secondary');

        secNotice.style.display = tab === 'notice' ? 'block' : 'none';
        secOrders.style.display = tab === 'orders' ? 'block' : 'none';

      function formatMonthDay(dateStr) {
        if (!dateStr) return '';
        let parts = dateStr.split(/[-/]/);
        if (parts.length >= 3) {
          return parts[1] + '/' + parts[2];
        }
        if (parts.length === 2) {
          return parts[0] + '/' + parts[1];
        }
        return dateStr;
      }

      function loadData() {
        google.script.run.withSuccessHandler(res => {
          let rawRows = res.rows || [];
          availableDates = res.dates || [];

          globalAllRows = rawRows.map(row => {
            let boolChecked = (row.orderStatus === '已購買');
            let pickupChecked = (row.orderStatus === '已取');
            return { ...row, checked: boolChecked, pickupChecked: pickupChecked };
          });

          if (availableDates.length > 0) {
            if (!selectedDate || !availableDates.includes(selectedDate)) {
              selectedDate = availableDates[0];
            }
            renderDateButtons();
          }

          if (currentTab === 'purchase' || currentTab === 'online') renderTableForSelectedDate();
          else if (currentTab === 'pickup') renderPickupTable();
          else if (currentTab === 'notice') {
            renderNoticeDateButtons();
            loadNotificationData(noticeSelectedDate);
          }
          else if (currentTab === 'orders') renderOrdersTable();
          else if (currentTab === 'history') renderHistoryTable();
        }).getPurchaseData();
      }

      function loadProductData() {
        google.script.run.withSuccessHandler(res => {
          globalProducts = res.products || [];
          if (currentTab === 'product-db') renderProductDbTable();
          if (currentTab === 'orders') renderOrdersTable();
        }).getProductsData();
      }

      function renderNoticeDateButtons() {
        let container = document.getElementById('notice-date-buttons-bar');
        if (!container) return;

        if (!availableDates || availableDates.length === 0) {
          container.innerHTML = '<span class="text-muted small">目前無採購日期資料</span>';
          return;
        }

        if (!noticeSelectedDate || !availableDates.includes(noticeSelectedDate)) {
          noticeSelectedDate = availableDates[0];
        }

        let html = '';
        availableDates.forEach(d => {
          let displayDate = formatMonthDay(d);
          let isActive = (d === noticeSelectedDate);
          let btnClass = isActive ? 'btn btn-primary date-btn shadow-sm' : 'btn btn-outline-secondary date-btn';
          html += `<button type="button" class="${btnClass}" onclick="selectNoticeDate('${d}')">${displayDate}</button>`;
        });
        
        container.innerHTML = html;
      }

      function selectNoticeDate(dateStr) {
        noticeSelectedDate = dateStr;
        renderNoticeDateButtons();
        loadNotificationData(noticeSelectedDate);
      }

      function loadNotificationData(targetDate) {
        let container = document.getElementById('notice-container');
        container.innerHTML = '<div class="text-center p-4 text-muted w-100">資料產出中...</div>';

        google.script.run.withSuccessHandler(res => {
          if (!res.success) {
            container.innerHTML = '<div class="text-center p-4 text-danger w-100">載入失敗！</div>';
            return;
          }

          let channelData = res.data || {};
          let targetChannels = ['LINE', 'LINE(私)', 'IG', 'FB', 'Threads'];
          let html = '';

          targetChannels.forEach(ch => {
            let usersMap = channelData[ch] || {};
            let userKeys = Object.keys(usersMap);

            html += `<div class="notice-channel-col">
              <div class="d-flex justify-content-between align-items-center bg-dark text-white p-2 mb-2 rounded">
                <span class="fw-bold">${ch}</span>
                ${userKeys.length > 0 ? `<button type="button" class="btn btn-outline-light btn-sm py-0 px-2" style="font-size: 0.75rem;" onclick="copyColumnNotices(this)">📋 複製 ${ch} 全部</button>` : ''}
              </div>`;

            if (userKeys.length === 0) {
              html += `<div class="notice-card text-muted text-center py-4">無 ${ch} 待取訂單</div>`;
            } else {
              userKeys.forEach(userKey => {
                let uData = usersMap[userKey];
                let headerPrefix = '';
                if (ch === 'FB') headerPrefix = '📍 (FB) ';
                else if (ch === 'LINE' || ch === 'LINE(私)' || ch === 'IG' || ch === 'Threads') headerPrefix = '@';

                let headerLine = `${headerPrefix}${uData.userName}`;
                let itemLines = [];

                uData.items.forEach(it => {
                  if (it.isOutOfStock || it.orderStatus === '缺貨') {
                    itemLines.push(`⚠️ ${it.name} ${it.qty}項 (缺貨)`);
                  } else {
                    itemLines.push(`🔹 ${it.name} ${it.qty}項 $${it.subtotal}`);
                  }
                });

                let totalAmount = uData.totalSubtotal + uData.totalFee;
                let summaryLine = `💰 商品金額 $${uData.totalSubtotal} + 代購費 $${uData.totalFee}\n🛒 總金額 $${totalAmount}`;

                let fullText = `${headerLine}\n${itemLines.join('\n')}\n${summaryLine}`;

                html += `<div class="notice-card shadow-sm position-relative">
                  <button type="button" class="btn btn-outline-secondary btn-sm position-absolute top-0 end-0 m-1 px-2 py-0" style="font-size: 0.7rem;" onclick="copyNoticeText(this)">📋 複製</button>
                  <div class="notice-text-content">${escapeHtml(fullText)}</div>
                </div>`;
              });
            }

            html += `</div>`;
          });

          container.innerHTML = html;
        }).getNotificationData(targetDate || '');
      }

      function escapeHtml(str) {
        return str.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
      }

      function copyNoticeText(btn) {
        let card = btn.closest('.notice-card');
        let text = card.querySelector('.notice-text-content').innerText;
        navigator.clipboard.writeText(text).then(() => {
          let originalText = btn.innerText;
          btn.innerText = '✅ 已複製';
          btn.classList.replace('btn-outline-secondary', 'btn-success');
          setTimeout(() => {
            btn.innerText = originalText;
            btn.classList.replace('btn-success', 'btn-outline-secondary');
          }, 1500);
        });
      }

      function copyColumnNotices(btn) {
        let col = btn.closest('.notice-channel-col');
        let cards = col.querySelectorAll('.notice-text-content');
        let allTexts = [];
        cards.forEach(c => {
          allTexts.push(c.innerText);
        });
        let combinedText = allTexts.join('\n\n--------------------\n\n');

        navigator.clipboard.writeText(combinedText).then(() => {
          let originalText = btn.innerText;
          btn.innerText = '✅ 已全部複製';
          btn.classList.replace('btn-outline-light', 'btn-success');
          setTimeout(() => {
            btn.innerText = originalText;
            btn.classList.replace('btn-success', 'btn-outline-light');
          }, 1500);
        });
      }

      function renderTableForSelectedDate() {
        let isOnlineMode = (currentTab === 'online');
        let tableContainerId = isOnlineMode ? 'online-purchase-table' : 'purchase-table';
        let expectedCountId = isOnlineMode ? 'online-expected-count' : 'expected-count';
        let actualCountId = isOnlineMode ? 'online-actual-count' : 'actual-count';
        let categorySummaryBarId = isOnlineMode ? 'online-category-summary-bar' : 'category-summary-bar';

        let formattedSelectedDate = formatMonthDay(selectedDate);

        let filteredRows = globalAllRows.filter(row => {
          let rowPurchaseDate = formatMonthDay(row.purchaseDate || row.date);
          let isStatusMatch = (row.orderStatus === '未買' || row.orderStatus === '已購買' || row.orderStatus === '缺貨');
          if (!isStatusMatch) return false;
          
          if (isOnlineMode) {
            return row.shippingChannel === '線上代購';
          } else {
            return (rowPurchaseDate === formattedSelectedDate) && (row.shippingChannel !== '線上代購');
          }
        });

        if (filteredRows.length === 0) {
          document.getElementById(tableContainerId).innerHTML = `<div class="text-center p-4 text-muted">目前無${isOnlineMode ? '線上代購' : '現場採購'}項目 (${formattedSelectedDate})。</div>`;
          document.getElementById(expectedCountId).innerText = 0;
          document.getElementById(actualCountId).innerText = 0;
          document.getElementById(categorySummaryBarId).innerHTML = '<span class="text-muted small">無分類資料</span>';
          currentGroupedRows = [];
          return;
        }

        let totalExpected = filteredRows.reduce((sum, row) => sum + Number(row.qty || 0), 0);
        document.getElementById(expectedCountId).innerText = totalExpected;

        let categoryMap = {};
        filteredRows.forEach(row => {
          let cat = row.category || '未分類';
          let qty = Number(row.qty || 0);
          categoryMap[cat] = (categoryMap[cat] || 0) + qty;
        });

        let catHtml = '';
        categoryOptions.forEach(cat => {
          if (categoryMap[cat]) {
            catHtml += `<div class="category-card">
              <span class="text-muted d-block" style="font-size: 0.75rem;">${cat}</span>
              <span class="fw-bold text-dark" style="font-size: 1rem;">${categoryMap[cat]} 件</span>
            </div>`;
          }
        });
        document.getElementById(categorySummaryBarId).innerHTML = catHtml;

        let productGroupMap = {};
        filteredRows.forEach(row => {
          let key = (row.code ? row.code.trim().toLowerCase() : '') + '|' + (row.name ? row.name.trim() : '');
          if (!productGroupMap[key]) {
            productGroupMap[key] = {
              code: row.code,
              name: row.name,
              category: row.category,
              price: row.price,
              totalQty: 0,
              subRows: [],
              userQtyMap: {},
              isOutOfStock: (row.orderStatus === '缺貨')
            };
          }
          let group = productGroupMap[key];
          let q = Number(row.qty || 0);
          group.totalQty += q;
          group.subRows.push(row);

          let userKey = row.user || '未具名';
          group.userQtyMap[userKey] = (group.userQtyMap[userKey] || 0) + q;
        });

        let groupedArray = Object.values(productGroupMap);

        groupedArray.sort((a, b) => {
          let indexA = categoryOptions.indexOf(a.category);
          let indexB = categoryOptions.indexOf(b.category);
          if (indexA === -1) indexA = 999;
          if (indexB === -1) indexB = 999;

          if (indexA !== indexB) {
            return indexA - indexB;
          }
          let nameA = (a.name || '').toString();
          let nameB = (b.name || '').toString();
          return nameA.localeCompare(nameB, 'zh-Hant');
        });

        currentGroupedRows = groupedArray;

        let html = '<table class="table table-striped table-hover mb-0 align-middle text-center purchase-large-table">';
        html += '<thead class="table-dark"><tr>';
        html += '<th style="width: 7%;">勾選</th>';
        html += '<th style="width: 7%;">缺貨</th>';
        html += '<th style="width: 7%;">數量</th>';
        html += '<th style="width: 14%;">單價</th>';
        html += '<th style="width: 16%;">商品代號</th>';
        html += '<th style="width: 23%; text-align: left;">商品名稱</th>';
        html += '<th style="width: 8%;">分類</th>';
        html += '<th style="width: 18%; text-align: left;">訂購人</th>';
        html += '</tr></thead><tbody>';

        currentGroupedRows.forEach((group, groupIdx) => {
          let allChecked = group.subRows.every(r => r.checked);
          let isCheckedStr = allChecked ? 'checked' : '';
          let isOosStr = group.isOutOfStock ? 'checked' : '';
          let category = group.category || '';
          let badgeClass = getCategoryBadgeClass(category);

          let userParts = [];
          for (let u in group.userQtyMap) {
            userParts.push(`${u} [${group.userQtyMap[u]}]`);
          }
          let userDisplayStr = userParts.join('、');
          let cleanCode = group.code ? group.code.replace(/^#+/, '') : '';

          html += `<tr id="group-row-${groupIdx}" class="${allChecked ? 'table-success text-decoration-line-through' : (group.isOutOfStock ? 'table-warning' : '')}">
            <td><input type="checkbox" id="group-chk-${groupIdx}" class="form-check-input" ${isCheckedStr} onchange="toggleGroupRowUI(${groupIdx})"></td>
            <td><input type="checkbox" id="group-oos-${groupIdx}" class="form-check-input bg-warning border-dark" ${isOosStr} onchange="toggleGroupOosUI(${groupIdx})"></td>
            <td class="fw-bold text-primary">${group.totalQty}</td>
            <td><input type="number" class="spreadsheet-input text-center text-danger fw-bold" value="${group.price || 0}" min="0" onchange="updateGroupPrice(${groupIdx}, this.value)"></td>
            <td><input type="text" class="spreadsheet-input text-center text-primary fw-bold" value="${cleanCode}" placeholder="代號" oninput="updateGroupCode(${groupIdx}, this)"></td>
            <td class="text-start fw-bold text-dark">${group.name || ''}</td>
            <td><span class="badge ${badgeClass}" style="font-size: 0.85rem;">${category}</span></td>
            <td class="text-start text-secondary fw-bold" style="font-size: 1rem;">${userDisplayStr}</td>
          </tr>`;
        });

        html += '</tbody></table>';
        document.getElementById(tableContainerId).innerHTML = html;
        
        calculateActualCount();
      }

      function updateGroupPrice(groupIdx, newPrice) {
        let group = currentGroupedRows[groupIdx];
        if (!group) return;
        let priceNum = Number(newPrice) || 0;
        group.price = priceNum;

        let updates = [];
        group.subRows.forEach(row => {
          row.price = priceNum;
          updates.push({ rowIndex: row.originalRowIndex, price: priceNum });
        });

        autoSaveGroupChanges(updates);
      }

      function updateGroupCode(groupIdx, inputEl) {
        let group = currentGroupedRows[groupIdx];
        if (!group) return;
        let codeVal = inputEl.value.trim();
        group.code = codeVal;

        let foundProduct = globalProducts.find(p => p.code.toLowerCase() === codeVal.toLowerCase());
        if (foundProduct) {
          group.name = foundProduct.name;
          group.category = foundProduct.category;
          group.price = foundProduct.price;
          renderTableForSelectedDate(); 
        }

        let updates = [];
        group.subRows.forEach(row => {
          row.code = codeVal;
          if (foundProduct) {
            row.name = foundProduct.name;
            row.category = foundProduct.category;
            row.price = foundProduct.price;
          }
          updates.push({ rowIndex: row.originalRowIndex, code: codeVal, name: row.name, category: row.category, price: row.price });
        });

        autoSaveGroupChanges(updates);
      }

      function autoSaveGroupChanges(updates) {
        let statusElId = (currentTab === 'online') ? 'online-sync-status' : 'sync-status';
        let statusEl = document.getElementById(statusElId);
        statusEl.innerText = "⏳ 儲存中...";

        google.script.run
          .withSuccessHandler(res => {
            if (res.success) {
              statusEl.innerText = "✨ 已自動儲存";
              setTimeout(() => { statusEl.innerText = ""; }, 1500);
            } else {
              statusEl.innerText = "❌ 儲存失敗";
            }
          })
          .batchUpdateRowStatus(updates);
      }

      function toggleGroupRowUI(groupIdx) {
        let group = currentGroupedRows[groupIdx];
        if (!group) return;

        let checkbox = document.getElementById(`group-chk-${groupIdx}`);
        let isChecked = checkbox.checked;

        if (isChecked && group.isOutOfStock) {
          group.isOutOfStock = false;
          let oosChk = document.getElementById(`group-oos-${groupIdx}`);
          if (oosChk) oosChk.checked = false;
        }

        let rowEl = document.getElementById(`group-row-${groupIdx}`);
        if (isChecked) {
          rowEl.className = 'table-success text-decoration-line-through';
        } else {
          rowEl.className = '';
        }

        let targetStatus = isChecked ? '已購買' : '未買';
        let updates = [];

        group.subRows.forEach(row => {
          row.checked = isChecked; // 💡 確保 subRow 的 checked 屬性同步更新
          row.orderStatus = targetStatus;
          updates.push({ rowIndex: row.originalRowIndex, targetStatus: targetStatus });
        });

        calculateActualCount();
        autoSaveGroupChanges(updates);
      }

      function toggleGroupOosUI(groupIdx) {
        let group = currentGroupedRows[groupIdx];
        if (!group) return;

        let oosChk = document.getElementById(`group-oos-${groupIdx}`);
        group.isOutOfStock = oosChk.checked;

        if (group.isOutOfStock) {
          group.subRows.forEach(r => {
            r.checked = false;
            r.orderStatus = '缺貨';
          });
          let mainChk = document.getElementById(`group-chk-${groupIdx}`);
          if (mainChk) mainChk.checked = false;
        } else {
          group.subRows.forEach(r => {
            r.orderStatus = '未買';
          });
        }

        let rowEl = document.getElementById(`group-row-${groupIdx}`);
        if (group.isOutOfStock) {
          rowEl.className = 'table-warning';
        } else {
          rowEl.className = '';
        }

        let updates = [];
        group.subRows.forEach(row => {
          updates.push({
            rowIndex: row.originalRowIndex,
            targetStatus: row.orderStatus
          });
        });

        calculateActualCount();
        autoSaveGroupChanges(updates);
      }

      function calculateActualCount() {
        let isOnlineMode = (currentTab === 'online');
        let actualCountId = isOnlineMode ? 'online-actual-count' : 'actual-count';

        let filteredRows = globalAllRows.filter(row => {
          let isStatusMatch = isOnlineMode ? (row.orderStatus === '未買' || row.orderStatus === '已購買') : (row.date === selectedDate && (row.orderStatus === '未買' || row.orderStatus === '已購買'));
          if (!isStatusMatch) return false;
          return isOnlineMode ? (row.shippingChannel === '線上代購') : (row.shippingChannel !== '線上代購');
        });

        let actual = filteredRows.filter(r => r.orderStatus === '已購買').reduce((sum, r) => sum + Number(r.qty || 0), 0);
        document.getElementById(actualCountId).innerText = actual;
      }

      // 💡 直接檢查畫面上所有群組勾選框，確保 100% 抓取勾選項目
      function markAsPurchasedUnpicked() {
        let isOnlineMode = (currentTab === 'online');
        let purchasedUpdates = [];
        let oosUpdates = [];

        currentGroupedRows.forEach((group, groupIdx) => {
          let chkEl = document.getElementById(`group-chk-${groupIdx}`);
          let oosEl = document.getElementById(`group-oos-${groupIdx}`);
          
          let isChecked = chkEl ? chkEl.checked : false;
          let isOos = oosEl ? oosEl.checked : group.isOutOfStock;

          if (isOos) {
            group.subRows.forEach(row => {
              oosUpdates.push({
                rowIndex: row.originalRowIndex,
                isOutofStockToUnbought: true
              });
            });
          } else if (isChecked) {
            group.subRows.forEach(row => {
              row.orderStatus = '未取';
              purchasedUpdates.push({
                rowIndex: row.originalRowIndex,
                targetStatus: '未取'
              });
            });
          }
        });

        let totalCount = purchasedUpdates.length + oosUpdates.length;
        if (totalCount === 0) {
          alert('請先勾選已完成購買的商品，或是勾選缺貨的商品！');
          return;
        }

        let msg = `確認處理：\n`;
        if (purchasedUpdates.length > 0) msg += `✅ ${isOnlineMode ? '訂購完成' : '購買完成'}（轉入未取）：${purchasedUpdates.length} 筆\n`;
        if (oosUpdates.length > 0) msg += `⚠️ 缺貨（遞延至下個採購日並改回未買）：${oosUpdates.length} 筆\n`;

        if (!confirm(msg)) return;

        let updates = purchasedUpdates.concat(oosUpdates);
        let statusElId = isOnlineMode ? 'online-sync-status' : 'sync-status';
        let statusEl = document.getElementById(statusElId);
        statusEl.innerText = "⏳ 處理中...";

        google.script.run
          .withSuccessHandler(res => {
            if (res.success) {
              statusEl.innerText = "✨ 處理成功";
              loadData();
            } else {
              statusEl.innerText = "❌ 處理失敗";
            }
          })
          .batchUpdateRowStatus(updates);
      }

      function renderPickupTable() {
        let pickupRows = globalAllRows.filter(row => row.orderStatus === '未取');

        let unpickedUsers = [...new Set(pickupRows.map(r => r.user).filter(Boolean))].sort();
        let userSelectEl = document.getElementById('filterPickupUser');
        let currentSelectedUser = userSelectEl ? userSelectEl.value : '';

        if (userSelectEl) {
          let uHtml = '<option value="">👤 全部客戶 (待取)</option>';
          unpickedUsers.forEach(u => {
            uHtml += `<option value="${u}" ${currentSelectedUser === u ? 'selected' : ''}>${u}</option>`;
          });
          userSelectEl.innerHTML = uHtml;
        }

        if (currentSelectedUser) {
          pickupRows = pickupRows.filter(r => r.user === currentSelectedUser);
        }

        if (pickupRows.length === 0) {
          document.getElementById('pickup-table').innerHTML = '<div class="text-center p-4 text-muted fs-5">目前沒有符合條件的待取項目。</div>';
          return;
        }

        let groupMap = {};
        pickupRows.forEach(row => {
          let user = row.user || '未具名';
          if (!groupMap[user]) {
            groupMap[user] = { rows: [], totalQty: 0, activeAmount: 0 };
          }
          groupMap[user].rows.push(row);
          let rowAmt = Number(row.totalAmount || 0);
          groupMap[user].totalQty += Number(row.qty || 0);

          if (row.payStatus !== '已轉帳') {
            groupMap[user].activeAmount += rowAmt;
          }
        });

        let html = '<table class="table table-bordered table-hover mb-0 align-middle text-center pickup-large-table">';
        html += '<thead class="table-dark"><tr>';
        html += '<th style="width: 8%;">取貨勾選</th>';
        html += '<th style="width: 8%;">已轉帳</th>';
        html += '<th style="width: 44%; text-align: left;">細分商品名稱</th>';
        html += '<th style="width: 20%;">數量</th>';
        html += '<th style="width: 20%;">金額</th>';
        html += '</tr></thead><tbody>';

        for (let user in groupMap) {
          let group = groupMap[user];
          let groupAllChecked = group.rows.every(r => r.pickupChecked);
          let groupAllPaid = group.rows.every(r => r.payStatus === '已轉帳');

          html += `<tr class="customer-group-header bg-secondary-subtle" id="group-header-${user}">
            <td><input type="checkbox" class="form-check-input" ${groupAllChecked ? 'checked' : ''} onchange="toggleCustomerGroup('${user}', this.checked)"></td>
            <td><input type="checkbox" class="form-check-input" ${groupAllPaid ? 'checked' : ''} onchange="toggleCustomerGroupPay('${user}', this.checked)"></td>
            <td colspan="3" class="text-start text-dark py-2 ps-2">
              <span class="fw-bold me-2 fs-5">👤 ${user}</span>
              <span class="badge bg-dark me-1 fs-6">總計: ${group.totalQty} 件</span>
              <span class="badge bg-danger fs-6">總金額: $<span class="group-total-amount">${group.activeAmount.toLocaleString()}</span></span>
            </td>
          </tr>`;

          group.rows.forEach(row => {
            let globalIndex = globalAllRows.findIndex(r => r.originalRowIndex === row.originalRowIndex);
            let isCheckedStr = row.pickupChecked ? 'checked' : '';
            let isPaidStr = (row.payStatus === '已轉帳') ? 'checked' : '';
            let cleanCode = row.code ? row.code.replace(/^#+/, '') : '';

            html += `<tr id="pickup-row-${globalIndex}" class="${row.pickupChecked ? 'table-success text-decoration-line-through' : ''}">
              <td><input type="checkbox" id="pickup-chk-${globalIndex}" class="form-check-input" ${isCheckedStr} onchange="togglePickupUI(${globalIndex})"></td>
              <td><input type="checkbox" id="pickup-pay-${globalIndex}" class="form-check-input pay-checkbox" ${isPaidStr} onchange="togglePickupPayUI(${globalIndex})"></td>
              <td class="text-start fw-bold text-dark ps-2">
                <span class="text-secondary me-1">└</span> ${row.name || ''} 
                ${cleanCode ? `<span class="badge bg-light text-dark border ms-1 fs-6">#${cleanCode}</span>` : ''}
                ${row.shippingChannel === '線上代購' ? '<span class="badge bg-info text-dark ms-1 fs-6">線上</span>' : ''}
              </td>
              <td class="fw-bold text-primary fs-5">${row.qty || 0} 件</td>
              <td class="fw-bold text-danger fs-5">$${Number(row.totalAmount || 0).toLocaleString()}</td>
            </tr>`;
          });
        }

        html += '</tbody></table>';
        document.getElementById('pickup-table').innerHTML = html;
      }

      function togglePickupUI(globalIndex) {
        let row = globalAllRows[globalIndex];
        if (!row) return;
        let checkbox = document.getElementById(`pickup-chk-${globalIndex}`);
        row.pickupChecked = checkbox.checked;
        let rowEl = document.getElementById(`pickup-row-${globalIndex}`);
        if (checkbox.checked) {
          rowEl.classList.add('table-success', 'text-decoration-line-through');
        } else {
          rowEl.classList.remove('table-success', 'text-decoration-line-through');
        }
        
        autoSavePickupChanges(row);
      }

      function togglePickupPayUI(globalIndex) {
        let row = globalAllRows[globalIndex];
        if (!row) return;
        let checkbox = document.getElementById(`pickup-pay-${globalIndex}`);
        row.payStatus = checkbox.checked ? '已轉帳' : '未付款';
        
        updateGroupActiveAmount(row.user);
        autoSavePickupChanges(row);
      }

      function toggleCustomerGroup(user, isChecked) {
        let updates = [];
        globalAllRows.forEach(row => {
          if (row.orderStatus === '未取' && (row.user || '未具名') === user) {
            row.pickupChecked = isChecked;
            let globalIndex = globalAllRows.findIndex(r => r.originalRowIndex === row.originalRowIndex);
            if (globalIndex !== -1) {
              let chk = document.getElementById(`pickup-chk-${globalIndex}`);
              let rowEl = document.getElementById(`pickup-row-${globalIndex}`);
              if (chk) chk.checked = isChecked;
              if (rowEl) {
                if (isChecked) {
                  rowEl.classList.add('table-success', 'text-decoration-line-through');
                } else {
                  rowEl.classList.remove('table-success', 'text-decoration-line-through');
                }
              }
            }
            updates.push({ 
              rowIndex: row.originalRowIndex, 
              targetStatus: row.orderStatus,
              payStatus: row.payStatus || '未付款'
            });
          }
        });
        
        batchAutoSavePickup(updates);
      }

      function toggleCustomerGroupPay(user, isChecked) {
        let updates = [];
        globalAllRows.forEach(row => {
          if (row.orderStatus === '未取' && (row.user || '未具名') === user) {
            row.payStatus = isChecked ? '已轉帳' : '未付款';
            let globalIndex = globalAllRows.findIndex(r => r.originalRowIndex === row.originalRowIndex);
            if (globalIndex !== -1) {
              let chk = document.getElementById(`pickup-pay-${globalIndex}`);
              if (chk) chk.checked = isChecked;
            }
            updates.push({ 
              rowIndex: row.originalRowIndex, 
              targetStatus: row.orderStatus,
              payStatus: row.payStatus
            });
          }
        });

        updateGroupActiveAmount(user);
        batchAutoSavePickup(updates);
      }

      function updateGroupActiveAmount(user) {
        let userRows = globalAllRows.filter(row => row.orderStatus === '未取' && (row.user || '未具名') === user);
        let activeSum = userRows.filter(r => r.payStatus !== '已轉帳').reduce((sum, r) => sum + Number(r.totalAmount || 0), 0);
        
        let headerRow = document.getElementById(`group-header-${user}`);
        if (headerRow) {
          let amountSpan = headerRow.querySelector('.group-total-amount');
          if (amountSpan) {
            amountSpan.innerText = activeSum.toLocaleString();
          }
        }
      }

      function autoSavePickupChanges(row) {
        let statusEl = document.getElementById('pickup-sync-status');
        statusEl.innerText = "⏳ 自動儲存中...";

        let updateData = [{
          rowIndex: row.originalRowIndex,
          targetStatus: row.orderStatus,
          payStatus: row.payStatus || '未付款'
        }];

        google.script.run
          .withSuccessHandler(res => {
            if (res.success) {
              statusEl.innerText = "✨ 已自動儲存";
              setTimeout(() => { statusEl.innerText = ""; }, 1500);
            } else {
              statusEl.innerText = "❌ 儲存失敗";
            }
          })
          .batchUpdateRowStatus(updateData);
      }

      function markAsPickedUp() {
        let pickedRows = globalAllRows.filter(row => row.orderStatus === '未取' && row.pickupChecked);
        if (pickedRows.length === 0) {
          alert('請先勾選已完成取貨的項目！');
          return;
        }

        if (!confirm(`確定將已勾選的 ${pickedRows.length} 筆項目設為「已取」完成嗎？`)) return;

        let updates = [];
        pickedRows.forEach(row => {
          row.orderStatus = '已取';
          let currentPayStatus = row.payStatus || '已轉帳';
          updates.push({ 
            rowIndex: row.originalRowIndex, 
            targetStatus: '已取',
            payStatus: currentPayStatus
          });
        });

        let statusEl = document.getElementById('pickup-sync-status');
        statusEl.innerText = "⏳ 儲存中...";

        google.script.run
          .withSuccessHandler(res => {
            if (res.success) {
              statusEl.innerText = "✨ 儲存成功";
              loadData();
            } else {
              statusEl.innerText = "❌ 儲存失敗";
            }
          })
          .batchUpdateRowStatus(updates);
      }

      function renderOrdersTable() {
        let unboughtRows = globalAllRows.filter(row => row.orderStatus === '未買');

        let unboughtUsers = [...new Set(unboughtRows.map(r => r.user).filter(Boolean))].sort();
        let unboughtProducts = [...new Set(unboughtRows.map(r => r.name).filter(Boolean))].sort();

        let userSelectEl = document.getElementById('filterOrderUser');
        let productSelectEl = document.getElementById('filterOrderProduct');

        let currentSelectedUser = userSelectEl ? userSelectEl.value : '';
        let currentSelectedProduct = productSelectEl ? productSelectEl.value : '';

        if (userSelectEl) {
          let uHtml = '<option value="">👤 全部客戶 (未買)</option>';
          unboughtUsers.forEach(u => {
            uHtml += `<option value="${u}" ${currentSelectedUser === u ? 'selected' : ''}>${u}</option>`;
          });
          userSelectEl.innerHTML = uHtml;
        }

        if (productSelectEl) {
          let pHtml = '<option value="">🛍️ 全部商品 (未買)</option>';
          unboughtProducts.forEach(p => {
            pHtml += `<option value="${p}" ${currentSelectedProduct === p ? 'selected' : ''}>${p}</option>`;
          });
          productSelectEl.innerHTML = pHtml;
        }

        if (currentSelectedUser) {
          unboughtRows = unboughtRows.filter(r => r.user === currentSelectedUser);
        }
        if (currentSelectedProduct) {
          unboughtRows = unboughtRows.filter(r => r.name === currentSelectedProduct);
        }

        unboughtRows.sort((a, b) => {
          let pDateA = a.purchaseDate || '';
          let pDateB = b.purchaseDate || '';
          if (pDateA !== pDateB) return pDateA.localeCompare(pDateB);

          let chanA = a.channel || '';
          let chanB = b.channel || '';
          if (chanA !== chanB) return chanA.localeCompare(chanB, 'zh-Hant');

          let userA = a.user || '';
          let userB = b.user || '';
          if (userA !== userB) return userA.localeCompare(userB, 'zh-Hant');

          let nameA = a.name || '';
          let nameB = b.name || '';
          return nameA.localeCompare(nameB, 'zh-Hant');
        });

        let countMap = {};
        unboughtRows.forEach(row => {
          let key = `${(row.user||'').trim()}_${(row.name||'').trim()}_${row.qty||''}_${row.orderDate||''}`;
          countMap[key] = (countMap[key] || 0) + 1;
        });

        let uniqueUsers = [...new Set(globalAllRows.map(r => r.user).filter(Boolean))].sort();
        let uniqueProducts = [...new Set(globalProducts.map(p => p.name).filter(Boolean))].sort();

        let userDatalistHtml = '<datalist id="existingUsers">';
        uniqueUsers.forEach(u => { userDatalistHtml += `<option value="${u}">`; });
        userDatalistHtml += '</datalist>';

        let productDatalistHtml = '<datalist id="existingProducts">';
        uniqueProducts.forEach(p => { productDatalistHtml += `<option value="${p}">`; });
        productDatalistHtml += '</datalist>';

        let html = userDatalistHtml + productDatalistHtml;
        html += '<table class="table table-bordered table-hover mb-0 align-middle text-center orders-large-table" id="editableOrdersTable">';
        html += '<thead class="table-dark"><tr>';
        html += '<th style="min-width: 90px;">訂購日期</th>';
        html += '<th style="min-width: 90px;">採購日期</th>';
        html += '<th style="min-width: 100px;">來源管道</th>';
        html += '<th style="min-width: 110px;">客戶姓名</th>';
        html += '<th style="min-width: 110px;">商品代號</th>';
        html += '<th style="min-width: 180px; text-align: left;">商品名稱</th>';
        html += '<th style="min-width: 120px;">商品分類</th>';
        html += '<th style="min-width: 70px;">數量</th>';
        html += '<th style="min-width: 80px;">單價</th>';
        html += '<th style="min-width: 80px;">代購費</th>';
        html += '<th style="min-width: 80px;">小計</th>';
        html += '<th style="min-width: 100px;">應收/應付</th>';
        html += '<th style="min-width: 80px;">線上</th>';
        html += '<th style="min-width: 80px;">已轉帳</th>';
        html += '<th style="min-width: 90px;">操作</th>';
        html += '</tr></thead><tbody>';

        if (unboughtRows.length === 0) {
          html += `<tr><td colspan="15" class="text-center p-4 text-muted">目前無符合條件的未買訂單。</td></tr>`;
        } else {
          unboughtRows.forEach((row) => {
            let globalIndex = globalAllRows.findIndex(r => r.originalRowIndex === row.originalRowIndex);
            let key = `${(row.user||'').trim()}_${(row.name||'').trim()}_${row.qty||''}_${row.orderDate||''}`;
            let isDuplicate = countMap[key] > 1;

            html += generateOrderRowHtml(row, globalIndex, isDuplicate);
          });
        }

        html += '</tbody></table>';
        document.getElementById('orders-table').innerHTML = html;
      }

      function lookupProductByCode(inputEl) {
        let codeValue = inputEl.value.trim().toLowerCase();
        if (!codeValue) return;

        let foundProduct = globalProducts.find(p => p.code.toLowerCase() === codeValue);
        if (foundProduct) {
          let tr = inputEl.closest('tr');
          let nameInput = tr.querySelector('[name="name"]');
          let categorySelect = tr.querySelector('[name="category"]');
          let priceInput = tr.querySelector('[name="price"]');
          
          if (nameInput) nameInput.value = foundProduct.name;
          if (categorySelect && foundProduct.category) categorySelect.value = foundProduct.category;
          if (priceInput) priceInput.value = foundProduct.price;

          recalcRow(inputEl);
        }
      }

      function lookupProductByName(inputEl) {
        let nameValue = inputEl.value.trim().toLowerCase();
        if (!nameValue) return;

        let foundProduct = globalProducts.find(p => p.name.toLowerCase() === nameValue);
        if (foundProduct) {
          let tr = inputEl.closest('tr');
          let codeInput = tr.querySelector('[name="code"]');
          let categorySelect = tr.querySelector('[name="category"]');
          let priceInput = tr.querySelector('[name="price"]');
          
          if (codeInput) codeInput.value = foundProduct.code;
          if (categorySelect && foundProduct.category) categorySelect.value = foundProduct.category;
          if (priceInput) priceInput.value = foundProduct.price;

          recalcRow(inputEl);
        }
      }

      function generateOrderRowHtml(row, indexKey, isDuplicate = false) {
        let qtyVal = (row.qty !== undefined && row.qty !== null && row.qty !== '') ? row.qty : '';
        let priceVal = (row.price !== undefined && row.price !== null && row.price !== '') ? row.price : '';
        let feeVal = (row.fee !== undefined && row.fee !== null && row.fee !== '') ? row.fee : '';

        let subtotal = (qtyVal !== '' && priceVal !== '') ? Math.round(Number(qtyVal) * Number(priceVal)) : '';
        let totalAmount = (subtotal !== '' ? subtotal : 0) + (feeVal !== '' ? Number(feeVal) : 0);

        let isPaid = (row.payStatus === '已轉帳');
        let isOnline = (row.shippingChannel === '線上代購');

        let channelClass = getChannelColorClass(row.channel);
        let channelSelect = `<select class="spreadsheet-input ${channelClass}" name="channel" onchange="updateChannelStyle(this)">`;
        sourceOptions.forEach(opt => {
          let selected = (row.channel === opt) ? 'selected' : '';
          channelSelect += `<option value="${opt}" ${selected}>${opt}</option>`;
        });
        channelSelect += `</select>`;

        let categorySelect = `<select class="spreadsheet-input" name="category">`;
        categoryOptions.forEach(opt => {
          let selected = (row.category === opt) ? 'selected' : '';
          categorySelect += `<option value="${opt}" ${selected}>${opt}</option>`;
        });
        categorySelect += `</select>`;

        let rowClass = isDuplicate ? 'table-warning' : '';
        let duplicateBadge = isDuplicate ? `<span class="badge bg-danger me-1" style="font-size: 0.75rem;">⚠️ 重複</span>` : '';

        return `<tr data-index="${indexKey}" data-original-row="${row.originalRowIndex || ''}" class="${rowClass}">
          <td><input type="text" class="spreadsheet-input text-center" name="orderDate" value="${formatMonthDay(row.orderDate) || ''}" placeholder="MM/DD"></td>
          <td><input type="text" class="spreadsheet-input text-center" name="purchaseDate" value="${formatMonthDay(row.purchaseDate) || ''}" placeholder="MM/DD"></td>
          <td>${channelSelect}</td>
          <td><input type="text" class="spreadsheet-input fw-bold" name="user" value="${row.user || ''}" placeholder="姓名" list="existingUsers"></td>
          <td><input type="text" class="spreadsheet-input text-center fw-bold text-primary" name="code" value="${row.code || ''}" placeholder="代號" oninput="lookupProductByCode(this)"></td>
          <td>${duplicateBadge}<input type="text" class="spreadsheet-input fw-bold text-start d-inline-block" style="width: ${isDuplicate ? 'calc(100% - 65px)' : '100%'};" name="name" value="${row.name || ''}" placeholder="商品名稱" list="existingProducts" oninput="lookupProductByName(this)" onchange="lookupProductByName(this)"></td>
          <td>${categorySelect}</td>
          <td><input type="number" step="any" class="spreadsheet-input text-center" name="qty" value="${qtyVal}" placeholder="" min="0" oninput="recalcRow(this)"></td>
          <td><input type="number" class="spreadsheet-input text-center" name="price" value="${priceVal}" placeholder="" min="0" oninput="recalcRow(this)"></td>
          <td><input type="number" class="spreadsheet-input text-center" name="fee" value="${feeVal}" placeholder="" min="0" oninput="recalcRow(this)"></td>
          <td class="fw-bold subtotal-cell">${subtotal}</td>
          <td class="fw-bold text-danger total-cell">${totalAmount}</td>
          <td><input type="checkbox" class="form-check-input shipping-checkbox" ${isOnline ? 'checked' : ''}></td>
          <td><input type="checkbox" class="form-check-input pay-checkbox" ${isPaid ? 'checked' : ''} onchange="recalcRow(this)"></td>
          <td>
            <button type="button" class="btn btn-outline-success btn-sm py-1 px-1 me-1" title="複製此筆訂單" onclick="duplicateOrderRow(this)"><i class="bi bi-files"></i></button>
            <button type="button" class="btn btn-outline-danger btn-sm py-1 px-1" title="刪除" onclick="removeOrderRow(this)"><i class="bi bi-trash"></i></button>
          </td>
        </tr>`;
      }

      function updateChannelStyle(selectEl) {
        selectEl.className = 'spreadsheet-input ' + getChannelColorClass(selectEl.value);
      }

      function recalcRow(el) {
        let tr = el.closest('tr');
        let qtyStr = tr.querySelector('[name="qty"]').value;
        let priceStr = tr.querySelector('[name="price"]').value;
        let feeStr = tr.querySelector('[name="fee"]').value;

        let qty = qtyStr !== '' ? Number(qtyStr) : 0;
        let price = priceStr !== '' ? Number(priceStr) : 0;
        let fee = feeStr !== '' ? Number(feeStr) : 0;

        let subtotal = (qtyStr !== '' && priceStr !== '') ? Math.round(qty * price) : '';
        let totalAmount = (subtotal !== '' ? subtotal : 0) + fee;

        tr.querySelector('.subtotal-cell').innerText = subtotal;
        tr.querySelector('.total-cell').innerText = totalAmount;
      }

      let isAddingOrder = false;
      function addNewOrderRow() {
        if (isAddingOrder) return;
        isAddingOrder = true;
        let btn = document.getElementById('btnAddOrder');
        if (btn) btn.disabled = true;

        let tbody = document.querySelector('#editableOrdersTable tbody');
        if (!tbody) {
          renderOrdersTable();
          tbody = document.querySelector('#editableOrdersTable tbody');
        }
        
        let now = new Date();
        let month = String(now.getMonth() + 1).padStart(2, '0');
        let day = String(now.getDate()).padStart(2, '0');
        let todayStr = month + '/' + day;

        let emptyRow = { 
          orderDate: todayStr, 
          purchaseDate: '', 
          channel: 'LINE', 
          user: '', 
          code: '', 
          name: '', 
          category: '生活用品', 
          qty: '', 
          price: '', 
          fee: '', 
          payStatus: '未付款', 
          shippingChannel: '' 
        };
        
        let tempIndex = 'new_' + Date.now();
        tbody.insertAdjacentHTML('afterbegin', generateOrderRowHtml(emptyRow, tempIndex, false));

        setTimeout(() => {
          isAddingOrder = false;
          if (btn) btn.disabled = false;
        }, 400);
      }

      function removeOrderRow(btn) {
        let tr = btn.closest('tr');
        let originalRowIndex = tr.getAttribute('data-original-row');

        if (!originalRowIndex) {
          // 如果是剛新增還沒存檔的列，直接在畫面上移除
          tr.remove();
          return;
        }

        if (!confirm('確定要從 Google 試算表中永久刪除這筆訂單嗎？')) return;
        
        let statusEl = document.getElementById('orders-sync-status');
        statusEl.innerText = "⏳ 試算表刪除中...";

        // 先把畫面的這一列移除保持流暢
        tr.remove();

        let payload = {
          rowIndex: Number(originalRowIndex)
        };

        google.script.run
          .withSuccessHandler(res => {
            if (res.success) {
              statusEl.innerText = "✨ 試算表刪除成功";
              // 稍微延遲後重新載入，確保試算表刪除動作完全生效

              statusEl.innerText = "❌ 刪除失敗: " + res.message;
              loadData(); // 失敗則還原畫面
            }
          })
          .deleteOrderRow(payload);
      }

      function renderHistoryTable() {

            (row.code && row.code.toLowerCase().includes(keyword)) ||
            (row.user && row.user.toLowerCase().includes(keyword)) ||
            (row.name && row.name.toLowerCase().includes(keyword)) ||
            (row.category && row.category.toLowerCase().includes(keyword)) ||
            (row.orderStatus && row.orderStatus.toLowerCase().includes(keyword))
          );
        }

          document.getElementById('history-table').innerHTML = '<div class="text-center p-4 text-muted">無符合的歷史訂單紀錄。</div>';
          return;

        let uniqueProducts = [...new Set(globalProducts.map(p => p.name).filter(Boolean))].sort();

        userDatalistHtml += '</datalist>';

        html += '<thead class="table-dark"><tr>';
        html += '<th style="min-width: 90px;">訂購日期</th>';
        html += '<th style="min-width: 90px;">採購日期</th>';
        html += '<th style="min-width: 100px;">來源管道</th>';
        html += '<th style="min-width: 110px;">客戶姓名</th>';
        html += '<th style="min-width: 110px;">商品代號</th>';

        html += '<th style="min-width: 80px;">代購費</th>';
        html += '<th style="min-width: 80px;">小計</th>';
        html += '<th style="min-width: 100px;">應收/應付</th>';
        html += '<th style="min-width: 100px;">付款狀態</th>';
        html += '<th style="min-width: 110px;">訂單狀態</th>';
        html += '</tr></thead><tbody>';

        historyRows.forEach(row => {
          let globalIndex = globalAllRows.findIndex(r => r.originalRowIndex === row.originalRowIndex);
          let subtotal = Math.round((Number(row.qty) || 0) * (Number(row.price) || 0));
          let fee = Number(row.fee) || 0;

          let channelClass = getChannelColorClass(row.channel);
          let channelSelect = `<select class="spreadsheet-input ${channelClass}" name="channel" onchange="updateHistoryChannelStyle(this)">`;
          sourceOptions.forEach(opt => {
            let selected = (row.channel === opt) ? 'selected' : '';
            channelSelect += `<option value="${opt}" ${selected}>${opt}</option>`;
          });
          channelSelect += `</select>`;

          let categorySelect = `<select class="spreadsheet-input" name="category">`;
          categoryOptions.forEach(opt => {
            let selected = (row.category === opt) ? 'selected' : '';
            categorySelect += `<option value="${opt}" ${selected}>${opt}</option>`;
          });

          let isTransferred = (row.payStatus === '已轉帳');

      function lookupHistoryProductByCode(inputEl) {
        let codeValue = inputEl.value.trim().toLowerCase();
        if (!codeValue) return;

        let foundProduct = globalProducts.find(p => p.code.toLowerCase() === codeValue);
        if (foundProduct) {
          let tr = inputEl.closest('tr');
          let nameInput = tr.querySelector('[name="name"]');
          let categorySelect = tr.querySelector('[name="category"]');
          let priceInput = tr.querySelector('[name="price"]');
          
          if (nameInput) nameInput.value = foundProduct.name;
          if (categorySelect && foundProduct.category) categorySelect.value = foundProduct.category;
          if (priceInput) priceInput.value = foundProduct.price;

          recalcHistoryRow(inputEl);
        }
      }

      function lookupHistoryProductByName(inputEl) {
        let nameValue = inputEl.value.trim().toLowerCase();
        if (!nameValue) return;

        let foundProduct = globalProducts.find(p => p.name.toLowerCase() === nameValue);
        if (foundProduct) {
          let tr = inputEl.closest('tr');
          let codeInput = tr.querySelector('[name="code"]');
          let categorySelect = tr.querySelector('[name="category"]');
          let priceInput = tr.querySelector('[name="price"]');
          
          if (codeInput) codeInput.value = foundProduct.code;
          if (categorySelect && foundProduct.category) categorySelect.value = foundProduct.category;
          if (priceInput) priceInput.value = foundProduct.price;

          recalcHistoryRow(inputEl);
        }
      }

      function recalcHistoryRow(el) {
        let tr = el.closest('tr');
        let qtyStr = tr.querySelector('[name="qty"]').value;
        let priceStr = tr.querySelector('[name="price"]').value;
        let feeStr = tr.querySelector('[name="fee"]').value;

        let qty = qtyStr !== '' ? Number(qtyStr) : 0;
        let price = priceStr !== '' ? Number(priceStr) : 0;
        let fee = feeStr !== '' ? Number(feeStr) : 0;

        let subtotal = (qtyStr !== '' && priceStr !== '') ? Math.round(qty * price) : '';
        let totalAmount = (subtotal !== '' ? subtotal : 0) + fee;

        tr.querySelector('.subtotal-cell').innerText = subtotal;
        tr.querySelector('.total-cell').innerText = totalAmount;
      }
